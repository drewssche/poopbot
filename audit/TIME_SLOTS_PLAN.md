## 🔧 Технические заметки

### time_service.py

```python
def get_time_slot(dt: datetime, tz: str = "Europe/Minsk") -> str:
    """Определяет временной слот."""
    local_dt = dt.astimezone(pytz.timezone(tz))
    hour = local_dt.hour
    
    if 0 <= hour < 6:
        return "night"
    elif 6 <= hour < 12:
        return "morning"
    elif 12 <= hour < 18:
        return "afternoon"
    else:
        return "evening"

def get_slot_emoji(slot: str) -> str:
    return {"night": "🌙", "morning": "🌅", "afternoon": "☀️", "evening": "🌆"}[slot]

def get_slot_title(slot: str) -> str:
    return {
        "night": "Ночной серун",
        "morning": "Утренний жаворонок",
        "afternoon": "Дневной трудяга",
        "evening": "Вечерний философ"
    }[slot]
```
